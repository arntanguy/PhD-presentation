\begin{frame}
	\frametitle{Static/robust balance regions}
	Where should the CoM lie to achieve balance?
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			Newton-Euler: 
			\begin{equation}\label{Eq:NEforce}
				\sum_{i=1}^{n} \boldsymbol{f}_{i} + m\mathbf{g} = m\ddot{\CoM}
			\end{equation}
			\begin{equation}\label{Eq:NEmoments}
				\sum_{i=1}^{n} (\mathbf{r}_{i} - \CoM) \times \boldsymbol{f}_{i} + \boldsymbol{n}_{i} = \boldsymbol{\dot{L}}
			\end{equation}
		\end{column}
		
		\begin{column}{0.48\textwidth}
			Unilateral contacts with non--slippage condition add:
			\begin{equation}\label{Eq:Unilateral}
				\mathbf{u}_i^T \boldsymbol{f}_i \geq 0  
			\end{equation}
			\begin{equation}\label{Eq:Friction}
				\|(\mathbb{I}-\mathbf{u}_i \mathbf{u}_i^T)\boldsymbol{f}_i \| \leq \mu_i \mathbf{u}_i^T \boldsymbol{f}_i
			\end{equation}
		\end{column}
	\end{columns}
	
	\hfill \\
	The set of contact forces $\boldsymbol{f}$ and CoM positions $\CoM$ that respect equations \eqref{Eq:NEforce} to \eqref{Eq:Friction} balances the robot. If $\ddot{\CoM} = \mathbf{0}$ then the balance is said to be static.
\end{frame}

\begin{frame}
	\frametitle{Static/robust balance regions}

	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{equation}
				\sum_{i=1}^{n} \boldsymbol{f}_{i} + m\mathbf{g} = m\ddot{\CoM} \tag{\ref{Eq:NEforce}}
			\end{equation}
			\begin{equation}
				\sum_{i=1}^{n} (\mathbf{r}_{i} - \CoM) \times \boldsymbol{f}_{i} + \boldsymbol{n}_{i} = \boldsymbol{\dot{L}} \tag{\ref{Eq:NEmoments}}
			\end{equation}
			\begin{equation}
				\mathbf{u}_i^T \boldsymbol{f}_i \geq 0  \tag{\ref{Eq:Unilateral}}
			\end{equation}
			\begin{equation}
				\|(\mathbb{I}-\mathbf{u}_i \mathbf{u}_i^T)\boldsymbol{f}_i \| \leq \mu_i \mathbf{u}_i^T \boldsymbol{f}_i \tag{\ref{Eq:Friction}}
			\end{equation}
		\end{column}
		
		\begin{column}{0.48\textwidth}
