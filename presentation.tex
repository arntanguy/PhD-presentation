\begin{frame}
	\frametitle{Human assistance}
	\framesubtitle{How to assist?}
	Problem: how do we assist a person in their movements without imposing the full movement?
	
	\hfill \\

	Generally we only want to help them remain in balance.

	\hfill \\

	Luckily humanoid balance and its control is also our field of study !

\end{frame}

\begin{frame}
	\frametitle{Static/robust balance regions}
	Where should the CoM lie to achieve balance?
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			Newton-Euler: 
			\begin{equation}\label{Eq:NEforce}
				\sum_{i=1}^{n} \boldsymbol{f}_{i} + m\mathbf{g} = m\ddot{\CoM}
			\end{equation}
			\begin{equation}\label{Eq:NEmoments}
				\sum_{i=1}^{n} (\mathbf{r}_{i} - \CoM) \times \boldsymbol{f}_{i} + \boldsymbol{n}_{i} = \boldsymbol{\dot{L}}
			\end{equation}
		\end{column}
		
		\begin{column}{0.48\textwidth}
			Unilateral contacts with non--slippage condition add:
			\begin{equation}\label{Eq:Unilateral}
				\mathbf{u}_i^T \boldsymbol{f}_i \geq 0  
			\end{equation}
			\begin{equation}\label{Eq:Friction}
				\|(\mathbb{I}-\mathbf{u}_i \mathbf{u}_i^T)\boldsymbol{f}_i \| \leq \mu_i \mathbf{u}_i^T \boldsymbol{f}_i
			\end{equation}
		\end{column}
	\end{columns}
	
	\hfill \\
	The set of contact forces $\boldsymbol{f}$ and CoM positions $\CoM$ that respect equations \eqref{Eq:NEforce} to \eqref{Eq:Friction} balances the robot. If $\ddot{\CoM} = \mathbf{0}$ then the balance is said to be static.
\end{frame}

\begin{frame}
	\frametitle{Static/robust balance regions}

	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{equation}
				\sum_{i=1}^{n} \boldsymbol{f}_{i} + m\mathbf{g} = m\ddot{\CoM} \tag{\ref{Eq:NEforce}}
			\end{equation}
			\begin{equation}
				\sum_{i=1}^{n} (\mathbf{r}_{i} - \CoM) \times \boldsymbol{f}_{i} + \boldsymbol{n}_{i} = \boldsymbol{\dot{L}} \tag{\ref{Eq:NEmoments}}
			\end{equation}
			\begin{equation}
				\mathbf{u}_i^T \boldsymbol{f}_i \geq 0  \tag{\ref{Eq:Unilateral}}
			\end{equation}
			\begin{equation}
				\|(\mathbb{I}-\mathbf{u}_i \mathbf{u}_i^T)\boldsymbol{f}_i \| \leq \mu_i \mathbf{u}_i^T \boldsymbol{f}_i \tag{\ref{Eq:Friction}}
			\end{equation}
		\end{column}
		
		\begin{column}{0.48\textwidth}
			\centering
			\begin{tikzpicture}
				\node
				[label={[font=\scriptsize]below:Visualization of the robust static balance region in\footfullcite{audren2018tro}.}]
				{
					\includegraphics[height=3.5cm]{figures/figure_haudren.png}
				};
			\end{tikzpicture}
		\end{column}
	\end{columns}
	
	Audren et al. proposed to cap $\ddot{\CoM}$ instead of setting it to zero to achieve a {\bf robust} balance. A 3D region for the CoM was computed through iterative projection (SOCP).
	
\end{frame}

\begin{frame}
	\frametitle{Static/robust balance regions}
	Other works at LIRMM expanded this 3D robust region: Roux\footfullcite{roux2021icar} (friction cones linearization) and Samadi\footfullcite{samadi2021ral} (computation of a Chebychev ball).
	These were improvements to be able to compute this region in near real time (-100ms).
	
	\hfill \\

	Basically enforcing a CoM position inside of these ensures a force distribution for balance exists.
\end{frame}

\section{Dynamic balance region}
\begin{frame}
	\frametitle{Limitations of the previous region}
	
	\begin{itemize}
		\item Not designed for dynamic movements.
		\item Slow computation ($\backsim$60ms for 3 contacts).
		\item Not taking actuation capabilities into account (human weakness...).
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Towards a dynamic balance region}
	{\bf New idea:} using the DCM-VRP formulation, we get a linear relationship between the force space and the "position" space for our dynamic points.
	\\
	$\implies$ This means the force limits can define a new region to constraint our dynamic points.
	\begin{equation}\label{Eq:VRP forces}
		\VRP = \CoM - \frac{b^2}{m}\mathbf{F}_\text{ext} + b^2{\bf g}
	\end{equation}
\end{frame}

\begin{frame}
	\frametitle{Towards a dynamic balance region}
	\framesubtitle{Contact friction cones}
	We keep the non slippage and unilaterality constraints.

	\hfill \\

	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{equation}
				\mathbf{u}_i^T \boldsymbol{f}_i \geq 0  \tag{\ref{Eq:Unilateral}}
			\end{equation}
			\begin{equation}
				\|(\mathbb{I}-\mathbf{u}_i \mathbf{u}_i^T)\boldsymbol{f}_i \| \leq \mu_i \mathbf{u}_i^T \boldsymbol{f}_i \tag{\ref{Eq:Friction}}
			\end{equation}
		\end{column}
		\begin{column}{0.48\textwidth}
			They can be linearized as a set of hyperplanes or inequalities and form a polyhedral cone. They are easily used in a QP in this form.
			
			\hfill \\

			By definition they are not bounded so friction only is insufficient.
		\end{column}

	\end{columns}
	
\end{frame}


\begin{frame}
	\frametitle{Towards a dynamic balance region}
	\framesubtitle{Actuation polytopes}
	We can not apply infinite force along the contact normal: we want balance AND actuation consistency. \\
	Orsolino et al. proposed to adress this with a force polytope\footfullcite{orsolino2020tro} decoupling the EoM between the end-effectors.
	\begin{equation}
		\label{Forces set}
		\begin{aligned}
			\mathcal{A}_i = \lbrace &\textbf{f}_i \in \mathbb{R}^m | \exists \torque_i \in \mathbb{R}^{n_l} ~\text{s.t.}~
			& \transpose{\boldsymbol{M}}_{bi}  \dot{\textbf{\textit{v}}} + \boldsymbol{M}_i \ddot{\textbf{q}}_i + \textbf{C}(\textbf{q}_i, \dot{\textbf{q}}_i) = \torque_i + \transpose{\jacobianVector(\textbf{q}_i)} \textbf{f}_i, \\
			&~ \lowerBound{\torque}_i\leq\torque_i\leq\upperBound{\torque}_i \rbrace
		\end{aligned}
	\end{equation}
	They assumed a static case to simplify the problem and found the following constraints
	\begin{equation}
		\mathcal{A}_i = \lbrace \textbf{f}_i \in \mathbb{R}^m |~ \bs{g}(\textbf{q}_i) + \lowerBound{\torque}_i \leq \transpose{\jacobianVector(\textbf{q}_i)} \textbf{f}_i \leq \bs{g}(\textbf{q}_i)+\upperBound{\torque}_i \rbrace
	\end{equation}
\end{frame}

\begin{frame}
	\frametitle{Towards a dynamic balance region}
	\framesubtitle{Minkowsky sum}
	Unlike the litterature we have a {\bf linear} dependance to the force space: no IP needed to compute the balance region. \\
	We can use the constraints as H-descriptions of polytopes in $\mathbb{R}^3$, then compute their double description (HV) to compute their minkowsky sum\footfullcite{delos2014corr}. \\
	We get the set of feasible forces, that with a factor and a vertical offset is the set of feasible VRPs.
	\begin{equation}
		\VRP = \CoM - \frac{b^2}{m}\mathbf{F}_\text{ext} + b^2{\bf g} \tag{\ref{Eq:VRP forces}}
	\end{equation}
	
\end{frame}

\begin{frame}
	\frametitle{Drawbacks and improvements}
	\begin{columns}[T]
		\begin{column}{0.48\linewidth}
			Problems:
			\begin{itemize}
				\item Constraining the decoupled forces assumes the max can be used on every contact at the same time.
				\item Instead of assuming static case we could define zero contact acceleration. But this requires multiplying by $\jacobianVector$ which is not definite positive.
			\end{itemize}
		\end{column}
		\begin{column}{0.48\linewidth}
			Perspectives:
			\begin{itemize}
				\item This formulation allows to account for joint weakness of the human.
				\item The fast computations can be plugged to an MPC for feasiblity.
			\end{itemize}
		\end{column}
	\end{columns}

	
\end{frame}
