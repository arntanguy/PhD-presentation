\begin{frame}
	\frametitle{Robot modeling}
	\framesubtitle{Kinematic and dynamic models}
	Robots are modelled as a set of rigid bodies articulated by joints.
	\vfill
	\begin{columns}
		\begin{column}{0.5\textwidth}
			Joint space:
			\begin{equation}
				\jointState = 
				\begin{pmatrix}
				\jointState_b \\
				\jointState_j
				\end{pmatrix}
				\in \cross{\SE{3}}{\mathbb{R}^n}
			\end{equation}
		\end{column}
		\begin{column}{0.5\textwidth}
			Task space:
			\begin{equation}
				\dot{{\bf s}} = \mathbf{J}(\jointState)\jointVel
			\end{equation}
		\end{column}
	\end{columns}
	\vfill
	A floating base robot making N contacts with its environment respects the equation of motion:
	\begin{equation}
		\boldsymbol{M}(\jointState)\jointAcc + \textbf{C}(\jointState, \jointVel) = \sum_{i = 1}^{N}\transpose{\jacobianVector_{i}(\jointState)}\force_i + \transpose{\boldsymbol{S}}\torque
	\end{equation}
\end{frame}

\begin{frame}
	\frametitle{Robot modeling}
	\framesubtitle{Reduced order models}
	To simplify whole-body models, we often reduce the order and reason about the Center of Mass (CoM).
	The centroidal model gives the linear momentum
    \begin{equation}
      \label{Eq:NE_sum_forces}
      \sum_{i=1}^{n} \force_{i} + m\mathbf{g} = m\ddot{\CoM} 
    \end{equation}
    and the angular momentum
    \begin{equation}
      \label{Eq:NE_sum_moments}
      \sum_{i=1}^{n} (\mathbf{r}_{i} - \CoM) \times \force_{i} + \torque_{i} = \boldsymbol{\dot{L}}
    \end{equation}
\end{frame}

\begin{frame}
	\frametitle{Robot modeling}
	\framesubtitle{Reduced order models}
	Modeling movement in a linear way: 
    \begin{columns}
		\begin{column}{0.5\textwidth}
			Zero-Moment Point (ZMP):\\
			Accumulates sum of wrenches at a virtual application point.
			For a given wrench $\wrench_0 = \transpose{(\torque_0, \force_0)}$ and a surface normal $\mathbf{n}$
			\begin{equation}
				\label{Eq:ZMP}
				\ZMP = \frac{\mathbf{n}\times \torque_0}{\force_0.\mathbf{n}}
			\end{equation}
		\end{column}
		\begin{column}{0.5\textwidth}
			Linear Inverted Pendulum Mode (LIPM):\\
			Two assumptions:
			\begin{enumerate}
				\item $\boldsymbol{\dot{L}}=0$
				\item Constant height
			\end{enumerate}
			
			In this mode
			\begin{equation}
				\label{Eq:LIPM}
				\ddot{\CoM} = \omega^2(\CoM - \ZMP)
			\end{equation}
			and $\omega^2 = \frac{g}{h}$ the natural frequency of the pendulum ZMP-CoM
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{Balance and feasiblity}
	\framesubtitle{3D regions}
	What about when the contacts are not coplanar? Where should the CoM lie for balance?
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			Newton-Euler: 
			\begin{equation}\label{Eq:NEforce}
				\sum_{i=1}^{n} \force_{i} + m\mathbf{g} = m\ddot{\CoM}
			\end{equation}
			\begin{equation}\label{Eq:NEmoments}
				\sum_{i=1}^{n} (\mathbf{r}_{i} - \CoM) \times \force_{i} + \torque_{i} = \boldsymbol{\dot{L}}
			\end{equation}
		\end{column}
		
		\begin{column}{0.48\textwidth}
			Unilateral contacts with non--slippage condition (friction cones) add:
			\begin{equation}\label{Eq:Unilateral}
				\mathbf{u}_i^T \force_i \geq 0  
			\end{equation}
			\begin{equation}\label{Eq:Friction}
				\|(\mathbb{I}-\mathbf{u}_i \mathbf{u}_i^T)\force_i \| \leq \mu_i \mathbf{u}_i^T \force_i
			\end{equation}
		\end{column}
	\end{columns}
	
	\hfill \\
	The set of contact wrenches $\wrench = (\torque,\force)$ and CoM positions $\CoM$ that respect equations \eqref{Eq:NEforce} to \eqref{Eq:Friction} balances the robot. If $\ddot{\CoM} = \mathbf{0}$ then the balance is said to be static.
\end{frame}

\begin{frame}
	\frametitle{Balance and feasiblity}
	\framesubtitle{3D regions}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{equation}
				\sum_{i=1}^{n} \force_{i} + m\mathbf{g} = m\ddot{\CoM} \tag{\ref{Eq:NEforce}}
			\end{equation}
			\begin{equation}
				\sum_{i=1}^{n} (\mathbf{r}_{i} - \CoM) \times \force_{i} + \torque_{i} = \boldsymbol{\dot{L}} \tag{\ref{Eq:NEmoments}}
			\end{equation}
			\begin{equation}
				\mathbf{u}_i^T \force_i \geq 0  \tag{\ref{Eq:Unilateral}}
			\end{equation}
			\begin{equation}
				\|(\mathbb{I}-\mathbf{u}_i \mathbf{u}_i^T)\force_i \| \leq \mu_i \mathbf{u}_i^T \force_i \tag{\ref{Eq:Friction}}
			\end{equation}
		\end{column}
		
		\begin{column}{0.48\textwidth}
			\centering
			\begin{tikzpicture}
				\node
				[label={[font=\scriptsize]below:Visualization of the robust static balance region in\footfullcite{audren2018tro}.}]
				{
					\includegraphics[height=3.5cm]{figures/figure_haudren.png}
				};
			\end{tikzpicture}
		\end{column}
	\end{columns}
	
	Audren et al. proposed to cap $\ddot{\CoM}$ instead of setting it to zero to achieve a {\bf robust} balance. A 3D region for the CoM was computed through iterative projection (SOCP).
	
\end{frame}

\begin{frame}
	\frametitle{Balance and feasiblity}
	\framesubtitle{3D regions}
	Other works at LIRMM expanded this 3D robust region: Roux\footfullcite{roux2021icar} (friction cones linearization) and Samadi\footfullcite{samadi2021ral} (computation of a Chebychev ball).
	These were improvements to be able to compute this region in near real time (-100ms).
	
	\hfill \\

	Basically enforcing a CoM position inside of these ensures a force distribution for balance exists.
\end{frame}

\begin{frame}
	\frametitle{Quadratic programming for robotics}
	For real-time robotic control we solve a Quadratic Problem (QP) at each control iteration.
	\begin{equation}
      \begin{aligned}
        \min_{\boldsymbol{x} \in \mathbb{R}^n} \quad & \frac{1}{2} ||\boldsymbol{A}\boldsymbol{x} + \boldsymbol{b}||^2_{\boldsymbol{W}} \\
        \textrm{s.t.} \quad & \boldsymbol{C}_{\text{ineq}} \boldsymbol{x} \leq \boldsymbol{d}_{\text{ineq}} \\
                            & \boldsymbol{C}_{\text{eq}} \boldsymbol{x} = \boldsymbol{d}_{\text{eq}} 
      \end{aligned}
    \end{equation}
	We minimize tasks errors (posture, movement...) under constraints (contacts, collisions...).
\end{frame}

\begin{frame}
	\frametitle{Incorporation in a multi-robot QP}
	\framesubtitle{Model actuation}
	The human model is actuated as a robot in the controller, retargeting the motion capture.

	The measured pose $\PluckerPose_d,$ velocity $\dot{\PluckerPose}_d$ and acceleration $\ddot{\PluckerPose}_d$ of each real human segment are given as task-space objectives for each body of the human model as follows:

	\begin{equation}
		\label{Eq:TransformTask}
		\begin{aligned}
			&\ddot{\PluckerPose} = \ddot{\PluckerPose}_d - \boldsymbol{K}_D\Delta\dot{\PluckerPose} - \boldsymbol{K}_P\Delta\PluckerPose \\
			& \mathrm{where} \ \ \ \ \Delta\PluckerPose = 
			\begin{bmatrix}
			\Delta\PluckerPose_A \\ 
			\Delta\PluckerPose_L 
			\end{bmatrix} = 
			\begin{bmatrix}
			\ln \left(\boldsymbol{E} {\boldsymbol{E}}_d^{\mathsf {T}}\right) \\
			\boldsymbol{r} - \boldsymbol{r}_d
			\end{bmatrix},
		\end{aligned}
	\end{equation}
\end{frame}

\begin{frame}
	\frametitle{Dynamic balance criterion}
	\framesubtitle{The Divergent Component of Motion}
	\begin{columns}
		\begin{column}{0.48\textwidth}
			\begin{equation*}
				\begin{aligned}
					\DCM=\CoM+{\frac{1}{\omega_{0}}}\dot{\CoM} \\
					\omega_{0} = \sqrt{\frac{g}{\Delta z}} \\
					\VRP = \CoM - \frac{\mathbf{F}_\text{ext}}{m\omega_{0}^{2}} + [0,0,\Delta z]^{T} \\
					\dot{\DCM}=\omega_{0}(\DCM-\VRP)
				\end{aligned}
			\end{equation*}
			3D DCM introduced by Englesberger\footfullcite{englsberger2015tro}.
		\end{column}
		\begin{column}{0.48\textwidth}

\begin{frame}
	\frametitle{Human assistance computation}
	\framesubtitle{DCM feedback}
	The reference VRP to apply to achieve the desired DCM dynamics is
	\vfill
	\begin{equation}
		\label{Eq:ref VRP computation}
		\VRP_r = \DCM - \frac{1}{\omega-\frac{\dot{\omega}}{\omega}}\left[ \dot{\DCM}_d + {\bf k}_{P}(\DCM_{d}-\DCM)\right]
	\end{equation}
\end{frame}

\begin{frame}
	\frametitle{Human assistance computation}
	\framesubtitle{Wrench computation}
	From the computed reference VRP $\VRP_r$ and the VRP formulations there are two ways of computing the missing assistive wrench
	\vfill
	\begin{columns}
		\begin{column}{0.5\textwidth}
			Measuring the wrench applied on the human and computing the missing force $\mathbf{F}_{r}$ at the CoM directly
			\begin{equation}
				\label{Eq:acceleration forces}
				\mathbf{F}_\text{ext} = \mathbf{F}_{r} + {\bf f}_\text{LF} + {\bf f}_\text{RF}
  			\end{equation}
			\begin{equation}
				\begin{aligned}
				\label{Eq:Robot hand forces}
				\mathbf{F}_\text{ext} &= m(\omega^{2}-\dot{\omega})(\CoM - \VRP_r) + m{\bf g} \\
				\mathbf{F}_{r} &= m(\omega^{2}-\dot{\omega})(\CoM - \VRP_r)  + m{\bf g} - {\bf f}_\text{LF} - {\bf f}_\text{RF} \\
				\end{aligned}
			\end{equation}
		\end{column}
		\begin{column}{0.5\textwidth}
			Measuring the acceleration of the human's CoM, and deducing the force from the missing acceleration $\ddot{\CoM}_e$
			\begin{equation}
				\begin{aligned}
				\label{Eq:Robot_hand_model}
				\ddot{\CoM}_d &= (\omega^{2}-\dot{\omega})(\CoM - \VRP_r) \\
				\ddot{\CoM}_e &= (\omega^{2}-\dot{\omega})(\CoM - \VRP_r) - \ddot{\CoM} \\
				\mathbf{F}_{r} &= m(\omega^{2}-\dot{\omega})(\CoM - \VRP_r) - m\ddot{\CoM} \\
				\end{aligned}
			\end{equation}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{Human assistance computation}
	\framesubtitle{Wrench distribution}
	\begin{equation}
		\begin{aligned}
		\min_{{\boldsymbol{w}}_\text{RH}, {\boldsymbol{w}}_\text{LH}} \quad & ||{\bf f}_\text{RH} + {\bf f}_\text{LH} - \mathbf{F}_{r}||^2 + ||{\boldsymbol{w}}_\text{RH}||^2 + ||{\boldsymbol{w}}_\text{LH}||^2 \\
		&+ ||({\bf r}_{\text{RH}} - \CoM) \times {\bf f}_{\text{RH}} + ({\bf r}_\text{LH} - \CoM) \times {\bf f}_\text{LH} ||^2\\  %+ ||w_{RH}||^2 + ||w_{LH}||^2 if min square
		\textrm{s.t.} \quad & {\boldsymbol{w}}_\text{RH}  \in \textrm{RH contact wrench cone} \\
							& {\boldsymbol{w}}_\text{LH}  \in \textrm{LH contact wrench cone} \\
							& f_{\text{RH}_z} + f_{\text{LH}_z} < \textrm{Max total force}  \\  
							& f_{\text{RH}_z} > \textrm{Min contact force}    \\
							& f_{\text{LH}_z} > \textrm{Min contact force}    \\
		\end{aligned}
	\end{equation}
	Contact wrench cones are formulated as inequalities for a rectangular contact surface\footfullcite{caron2015icra}
\end{frame}

\begin{frame}
	\frametitle{Wrench application on human}
	\framesubtitle{Impedance task}
	Position control:
	\begin{equation}
		\label{Eq:EndEff task}
		\Delta_\text{EF} = \boldsymbol{K}_P\Delta\PluckerPose + \boldsymbol{K}_D\Delta\dot{\PluckerPose} + \Delta\ddot{\PluckerPose}
	\end{equation}
	Force control:
	\begin{equation}
		\label{Eq:Impedance}
		\boldsymbol{M}\Delta\ddot{\PluckerPose} + \boldsymbol{D}\Delta\dot{\PluckerPose} + \boldsymbol{K}\Delta\PluckerPose = \boldsymbol{K}_f (\boldsymbol{w}_m - \boldsymbol{w}_d)
	\end{equation}
	\vfill
	Force control can be used to to compute a desired acceleration offset which can be added to the reference position control\footfullcite{bouyarmane2019tro}\footfullcite{murooka2022ral}.
\end{frame}

\begin{frame}

		\begin{column}{0.7\textwidth}
			\begin{equation}
				\label{Eq:EoM surface contacts}
				\boldsymbol{M}(\jointState)\jointAcc + \textbf{C}(\jointState, \jointVel) = \sum_{i = 1}^{N}\transpose{\jacobianVector_{i}(\jointState)}\wrench_i + \transpose{\boldsymbol{S}}\torque,
			\end{equation}

			\begin{equation}\label{Eq:target acc contact}
				^e\ddot{\mathbf{p}}_r = \jacobianVector(\jointState)\jointAcc + \dot{\jacobianVector}(\jointState)\jointVel.
			\end{equation}

			\begin{equation}
			\label{Eq:Wrench polytope Vrep}
			\begin{aligned}
				\jacobianVector\inverse{\boldsymbol{M}}\transpose{\jacobianVector}\wrench  &= \jacobianVector\inverse{\boldsymbol{M}}(\textbf{C} - \transpose{\boldsymbol{S}}\torque) + \jacobianVector\inverse{\boldsymbol{M}}\boldsymbol{M}\jointAcc \\
				&= \jacobianVector\inverse{\boldsymbol{M}}(\textbf{C} - \transpose{\boldsymbol{S}}\torque) - \dot{\jacobianVector}\jointVel + {^e\ddot{\mathbf{p}}_r} \\
				\inverse{\osInertia}\wrench &= \jacobianVector\inverse{\boldsymbol{M}}(\textbf{C} - \transpose{\boldsymbol{S}}\torque) - \dot{\jacobianVector}\jointVel + {^e\ddot{\mathbf{p}}_r} \\
				\wrench &= \osInertia\jacobianVector\inverse{\boldsymbol{M}}(\textbf{C} - \transpose{\boldsymbol{S}}\torque) - \osInertia\dot{\jacobianVector}\jointVel + \osInertia {^e\ddot{\mathbf{p}}_r} \\
				&= - \transpose{\overline{\jacobianVector}}\transpose{\boldsymbol{S}}\torque + \transpose{\overline{\jacobianVector}}\textbf{C} - \osInertia\dot{\jacobianVector}\jointVel + \osInertia {^e\ddot{\mathbf{p}}_r}

\begin{frame}
	\frametitle{Feasible Contact Wrench Polytopes}
	\framesubtitle{CoP constraints}
	In case of contact surfaces and not points, the application point of the force (Center of Pressure or CoP) can lie anywhere in the surface. This bounds the torque of the contact wrench. For rectangular surfaces this constraint is derived as \footfullcite{caron2015icra}
	\begin{align}
		|\tau^x| &\leq Y f^z \\
		|\tau^y| &\leq X f^z \\
		|\tau^z| &\leq \mu(X + Y)f^z - |Yf^x + \mu\tau^x| - |Xf^y + \mu\tau^y|
	\end{align}
	These inequalities can be written as 12 constraints depending on the friction coefficient and the surface dimensions.
\end{frame}

\begin{frame}
	\frametitle{Constrained DCM-VRP Control}
	\framesubtitle{DCM capturability}
	We wanted to design a region more suited to dynamic points like the DCM.
	\vfill
	If we remind the DCM-VRP formulations we can derive two propositions
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\begin{equation}
				\label{Eq:DCM-VRP}
				\dot{\DCM}=\omega(\DCM-\VRP)
			\end{equation}
			
			\begin{equation}
				\label{Eq:eCMP-VRP}
				\VRP = \CoM - \frac{\mathbf{F}_\text{ext}}{m\omega^2} + \transpose{[0,0,\Delta z]}
			\end{equation}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{proposition}
				Since the VRP repulses the DCM, see~\eqref{Eq:DCM-VRP}, the bounding set of the capturable DCM positions is the set of feasible VRPs.
			\end{proposition}
			\begin{proposition}
				Since the VRP is a linear function of the sum of external forces~\eqref{Eq:eCMP-VRP} then the set of feasible VRPs is the set of feasible sum of external forces, scaled and offset in the cartesian space.
			\end{proposition}
		\end{column}
	\end{columns}

\end{frame}

\begin{frame}
	\frametitle{Constrained DCM-VRP Control}
	\framesubtitle{Full VRP region}
	\begin{itemize}
		\item We can not simply operate a Minkowski sum of the contact polytopes to get the full set of feasible external forces/wrenches: coupling of the equation of motion.
		\item It could be computed using iterative projections~\footfullcite{audren2018tro} but it is costly.
	\end{itemize}
	
	If we assume we have this set (or a single contact), we also have lineraly the VRP region.
	Any command can be constrained in this region to ensure dynamic balance.
	\begin{equation}
		\begin{aligned}
		\min_{\VRP} \quad & ||\VRP - \VRP_d||^2  \\
		\textrm{s.t.} \quad & \VRP \in \cal{P}_\text{VRP} 
		\end{aligned}
	\end{equation}
\end{frame}

\begin{frame}
	\frametitle{Constrained DCM-VRP Control}
	\framesubtitle{Implicit bounding}
	\centering
	\begin{columns}
		\begin{column}{0.6\textwidth}
			\begin{align}
			\min_{\substack{\jointAcc, \torque, \wrench_i \\ 
							\jointAcc^{\text{virt}}, \torque^{\text{virt}}, \wrench_i^{\text{virt}}}} 
			\quad & ||\VRP - \VRP_{d}||^2  \label{Eq:DCM bounding objective} \\ 
			\textrm{s.t.} \quad & \boldsymbol{M}\jointAcc + \textbf{C} = \sum_{i = 1}^{N}\transpose{\jacobianVector_{i}}\wrench_i + \transpose{\boldsymbol{S}}\torque \\
								& \boldsymbol{M}\jointAcc^{\text{virt}} + \textbf{C} = \sum_{i = 1}^{N}\transpose{\jacobianVector_{i}}\wrench_i^{\text{virt}} + \transpose{\boldsymbol{S}}\torque^{\text{virt}}\\
								& \lowerBound{\torque} \leq \torque \leq \upperBound{\torque} \\
								& \lowerBound{\torque} \leq \torque^{\text{virt}} \leq \upperBound{\torque} \\
								& \VRP \in \cal{P}_\text{VRP} \label{Eq:DCM bounding VRP constraint} \\ 
								& \hat{\DCM} \in \cal{P}_\text{VRP} \label{Eq:DCM bounding dotDCM constraint}
			\end{align}
		\end{column}
		\begin{column}{0.4\textwidth}
			How do we express these constraints without computing the full VRP region?
			\begin{itemize}
				\item Solve the problem at the joint level.
				\item Introduce virtual joints, torques and contact wrenches variables to formulate $\hat{\DCM}$ as a VRP.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{Constrained DCM-VRP Control}
	\framesubtitle{Implicit bounding}
	\centering
	\begin{columns}
		\begin{column}{0.2\textwidth}
			Objective
			\begin{equation*}
				||\VRP - \VRP_{d}||^2 
			\end{equation*}
			becomes
			\begin{equation*}
				||\CoM - \frac{\mathbf{F}_\text{ext}}{m\omega^2} + \boldsymbol{\Delta}_z - \VRP_d||^2
			\end{equation*}
		\end{column}
		\begin{column}{0.2\textwidth}
			Constraint
			\begin{equation*}
				\VRP \in \cal{P}_\text{VRP}
			\end{equation*}
			becomes
			\begin{equation*}
				\wrench_i \in \cal{P}_\text{Feasible i}
			\end{equation*}
		\end{column}
		\begin{column}{0.4\textwidth}
			Constraint
			\begin{equation*}
				\hat{\DCM} \in \cal{P}_\text{VRP}
			\end{equation*}
			becomes
			\begin{equation*}
				\frac{\alpha\mathbf{F}_\text{ext}}{m\omega} + \frac{\mathbf{F}_\text{virt}}{m\omega^2} = (1 + \alpha\omega)(\CoM - \DCM + \boldsymbol{\Delta}_z)
			\end{equation*}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{Optimal DCM computation}
	Without a locomotion reference trajectory we want to just "remain balanced", i.e. compute automatically a target with the most robustness.
	\vfill
	The individual contact polytopes can be leveraged to determine this target according to the current posture.
\end{frame}

\begin{frame}
	\frametitle{Optimal DCM computation}
	\framesubtitle{Robust static DCM}
	\centering
	The point with the most margin within a polytope can be found as the Chebychev center.

	\begin{equation}
	\begin{aligned}
		\max_r \quad & r \\
		\textrm{s.t.} \quad & \boldsymbol{A} \ \boldsymbol{x} + r\norm{\boldsymbol{A}} \leq \boldsymbol{b}
	\end{aligned}
	\end{equation}
\end{frame}

\begin{frame}
	\frametitle{Optimal DCM computation}
	\framesubtitle{Robust static DCM}
	We want to find the DCM position for which we can remain statically balanced, i.e. for $\transpose{\wrench_\text{static}} = \begin{bmatrix} 0 & 0 & 0 & 0 & 0 & mg\end{bmatrix}$.
	\begin{equation} \label{Eq:Optimal CoM-Wrenches}
	\sum_{i = 1}^{N} \svaFT{i}{0}
	\begin{bmatrix}
		\torque_i \\
		\force_i
	\end{bmatrix}
	+ 
	\begin{bmatrix}
		\lfloor\mathbf{F}_\text{static}\rfloor_\times \CoM_\text{opti} \\
		\zeroMatrix
	\end{bmatrix}
	\!=\!
	\begin{bmatrix}
		\torque_\text{static} \\ 
		\mathbf{F}_\text{static}
	\end{bmatrix}
	\end{equation}
\end{frame}

\begin{frame}
	\frametitle{Optimal DCM computation}
	\framesubtitle{Optimal wrench distribution}
	The QP becomes
	\begin{equation} \label{Eq:CoM auto target}
	\begin{aligned}
		\min_{\substack{\wrench_i, r_i, \CoM_\text{opti}}} 
	\quad & -\sum_{i = 1}^{N}r_i + \sum_{i = 1}^{N}\wrench_i + \eqref{Eq:Optimal CoM-Wrenches} + k||\CoM_\text{opti} - \DCM_\text{prev}||^2  \\ 
	\textrm{s.t.} \quad & \boldsymbol{A}_i \ \wrench_i + r_i\norm{\boldsymbol{A}_i} \leq \boldsymbol{b}_i \\
						& -r_i \leq 0
	\end{aligned}
	\end{equation}
	The resulting optimal CoM position $ \CoM_\text{opti}$ is given as a target to the balancing task.
\end{frame}

\begin{frame}
	\frametitle{Optimal DCM computation}
	\framesubtitle{Automatic contact transitions}
	A secondary advantage of this formulation is that the contact polytopes $\mathcal{H}$-description can be scaled using a coefficient on the planes offsets.

	As the automatic balance framework relies on their margin to target the DCM this can be used for automatic contact transitions.
\end{frame}

\begin{frame}
	\frametitle{Drawbacks and perspectives}
	\begin{itemize}
		\item The coupling of the wrench polytopes limits their use without the whole-body QP.
		\item With studies on the average "torque limits" of impaired or elderly people, this exact framework could be used to better estimate the assitance needed.
		\item The fast computations could be plugged to an MPC for feasiblity.
	\end{itemize}
	
\end{frame}
